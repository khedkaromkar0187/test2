#include <stdio.h>

int prime(int n)
{
    int i;

    if(n < 2)
        return 0;

    for(i = 2; i < n; i++)
    {
        if(n % i == 0)
            return 0;
    }

    return 1;
}

void main()
{
    int start, end, i, sum = 0;

    printf("Enter starting number: ");
    scanf("%d", &start);

    printf("Enter ending number: ");
    scanf("%d", &end);

    for(i = start; i <= end; i++)
    {
        if(prime(i))
        {
            printf("%d ", i);
            sum = sum + i;
        }
    }

    printf("\nSum of prime numbers = %d", sum);
}

#include <stdio.h>

void replace(char str[], char ch, char symbol)
{
    int i;

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == ch)
        {
            str[i] = symbol;
        }
    }
}

void main()
{
    char str[50];
    char ch, symbol;

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Enter character to replace: ");
    scanf(" %c", &ch);

    printf("Enter special symbol: ");
    scanf(" %c", &symbol);

    replace(str, ch, symbol);

    printf("Updated string = %s", str);
}
